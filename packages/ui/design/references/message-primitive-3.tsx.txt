type MessageActionButtonProps = {

  state?: "Default" | "Hover";

export default function MessageActionButton({ className, iconSwap = null, state = "Default" }: MessageActionButtonProps) {
  const isHover = state === "Hover";
  return (
    <div className={className || "content-stretch flex items-center p-[var(--spacing-xxs,2px)] relative rounded-[var(--radius-xs,6px)]"} id={isHover ? "node-9206_591499" : "node-9206_591492"}>
      {state === "Default" &&
        (iconSwap || (
          <div className="relative shrink-0 size-[16px]" data-node-id="9206:591488" data-name="placeholder">
            <img alt="" className="absolute block inset-0 max-w-none size-full" src={imgPlaceholder} />
          </div>
        ))}
      {isHover &&
        (iconSwap || (

            <img alt="" className="absolute block inset-0 max-w-none size-full" src={imgPlaceholder1} />
          </div>
        ))}
      {isHover && (
        <div className="absolute bottom-[-9px] right-[-9px] size-[20px]" data-node-id="9206:591520" data-name="Cursor">
          <div className="absolute inset-[0_-0.61%_-5.6%_0]">
            <img alt="" className="block max-w-none size-full" src={imgCursor} />
          </div>
        </div>
      )}
