  className?: string;
  checked?: boolean;
  indeterminate?: "False";

function CheckboxBase({ className, checked = false, indeterminate = "False", size = "sm", state = "Default", type = "Checkbox" }: CheckboxBaseProps) {
  const isCheckedAndFalseAndSmAndRadioAndDefault = checked && indeterminate === "False" && size === "sm" && type === "Radio" && state === "Default";
  const isNotCheckedAndFalseAndSmAndRadioAndDefault = !checked && indeterminate === "False" && size === "sm" && type === "Radio" && state === "Default";
  return (
    <div className={className || `overflow-clip relative size-[16px] ${isCheckedAndFalseAndSmAndRadioAndDefault ? String.raw`bg-[var(--colors\/background\/bg-brand-solid,#079455)] rounded-[var(--radius-full,9999px)]` : isNotCheckedAndFalseAndSmAndRadioAndDefault ? String.raw`border border-[var(--colors\/border\/border-primary,#d5d7da)] border-solid rounded-[var(--radius-full,9999px)]` : String.raw`border border-[var(--colors\/border\/border-primary,#d5d7da)] border-solid rounded-[var(--radius-xs,6px)]`}`} id={isCheckedAndFalseAndSmAndRadioAndDefault ? "node-1100_552" : isNotCheckedAndFalseAndSmAndRadioAndDefault ? "node-1100_569" : "node-1097_63885"}>
