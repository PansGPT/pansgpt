  rightConnector?: boolean;
  state?: "Default" | "Hover" | "Disabled";

String.raw`bg-[var(--colors\/background\/bg-secondary_hover,#f5f5f5)]` : isSelectedAndDefault ? String.raw`bg-[var(--colors\/background\/bg-brand-solid,#079455)]` : isDefaultAndIsTodaysDateOrActive ? String.raw`bg-[var(--colors\/background\/bg-active,#fafafa)]` : ""}`} id={isSelectedAndHover ? "node-7365_58014" : isTodaysDateAndHover ? "node-7365_58000" : isDefaultAndHover ? "node-7365_57964" : isActiveAndHover ? "node-7365_57956" : isActiveAndDefault ? "node-7365_57947" : isSelectedAndDefault ? "node-7365_57944" : isTodaysDateAndDefault ? "node-7365_57930" : "node-7365_57925"}>
      {isTodaysDateOrSelectedOrActive && rightConnector && (

String.raw`font-[family-name:var(--font-family\/font-family-body,"Plus_Jakarta_Sans:Medium")] font-medium text-[color:var(--colors\/text\/text-secondary_hover,#252b37)]` : isSelectedAndIsDefaultOrHover ?
