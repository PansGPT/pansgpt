          </div>
        </div>
      </div>
    );
  }
  return (
    <div className={className || "border-[var(--colors\\/border\\/border-secondary,#e9eaeb)] border-b border-solid content-stretch flex gap-[var(--spacing-lg,12px)] h-[64px] items-center max-h-[64px] px-[var(--spacing-2xl,20px)] py-[var(--spacing-xl,16px)] relative w-[320px]"} data-node-id="9405:710420" data-name="Type=Text, Size=sm, State=Default">
      {leadAction && <TableCellLeadAction className="content-stretch flex items-start relative shrink-0" />}
      <div className="[word-break:break-word] content-stretch flex flex-col items-start leading-[var(--line-height\/text-sm,20px)] relative shrink-0 text-[length:var(--font-size\/text-sm,14px)] whitespace-nowrap" data-node-id="9405:710422" data-name="Text and supporting text">
        {leadText && (
          <p className="font-[family-name:var(--font-family\/font-family-body,'Plus_Jakarta_Sans:Medium')] font-medium relative shrink-0 text-[color:var(--colors\/text\/text-primary-\(900\),#181d27)]" data-node-id="9405:710423">
            Diepreye Ọ̀rẹ̀kọ́yà
          </p>
        )}
        {supportingText && (
          <p className="font-[family-name:var(--font-family\/font-family-body,'Plus_Jakarta_Sans:Regular')] font-normal relative shrink-0 text-[color:var(--colors\/text\/text-tertiary-\(600\),#535862)]" data-node-id="9405:710424">
            chioma@PansGPT.com
          </p>
        )}
      </div>
    </div>
  );
}
